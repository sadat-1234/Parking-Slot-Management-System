    int 21h

    mov dx, 10
    int 21h
    mov dx, 13
    int 21h

; FEATURE 1 SUPPORT: assigns the next Vehicle ID.
SET_ID MACRO
    mov al, GLOBAL_INDEX
    mov ah, 00h
    mov si, ax
    add al, 48
    mov VEHICLE_ID[si], al

    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h
    PRINT_MSG enter_id_complete
ENDM

; FEATURE 1 SUPPORT: stores the vehicle owner name in the array.
TAKE_NAME_INPUT MACRO
    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h
    mov ah, 9
    lea dx, enter_name
    int 21h

    mov cx, 20
    mov al, GLOBAL_INDEX
    mov bl, 20
    mul bl
    mov si, ax
    START_NAME:
        mov ah, 1
        int 21h
        cmp al, terminator
        je COMPLETE_NAME
        mov VEHICLE_OWNER[si], al
        add si, 1
    LOOP START_NAME
    COMPLETE_NAME:
    LOOP COMPLETE_NAME
    mov VEHICLE_OWNER[si], "$"
    PRINT_MSG enter_name_complete
ENDM

MAIN PROC

MOV AX, @DATA
MOV DS, AX

; MAIN MENU DISPATCH: options 1-6 jump to the six project features.
; The menu is printed every time control returns to MAIN_MENU so that
; the user can always see the available options.
MAIN_MENU:
    ; Re-display the main menu after every feature returns here.
    WELCOME_SCREEN

    mov ah, 1
    int 21h
    sub al, 30h
    mov command, al
    cmp command, 0h
    je END
    cmp command, 1
    je JUMP_ADD_VEHICLE
    cmp command, 2
    je JUMP_UPDATE_VEHICLE_INFO
    cmp command, 3
    je JUMP_SEARCH_VEHICLE
    cmp command, 4
    je JUMP_CALCULATE_FEE
    cmp command, 5
    je JUMP_DELETE_VEHICLE
    cmp command, 6
    je JUMP_CAPACITY_CHECK
    jmp MAIN_MENU

; ============================================================
; FEATURE 1: REGISTER A NEW VEHICLE
; MENU HANDLER
; ============================================================
JUMP_ADD_VEHICLE:
    mov al, GLOBAL_INDEX
    cmp al, MAX_CAPACITY
    jl  DO_ADD_VEHICLE
    PRINT_MSG add_blocked_msg
    jmp AFTER_ADD_VEHICLE
DO_ADD_VEHICLE:
    CALL ADD_VEHICLE
    mov command, 9h
    PRINT_MSG retuen_main_menu
    add GLOBAL_INDEX, 1
AFTER_ADD_VEHICLE:
    jmp MAIN_MENU

; ============================================================
; FEATURE 2: UPDATE VEHICLE INFORMATION
; MENU HANDLER
; ============================================================
JUMP_UPDATE_VEHICLE_INFO:
    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h
    PRINT_MSG update_vehicle_info_instruction

    mov ah, 1
    int 21h
    sub al, 30h
    mov GLOBAL_UPDATE_ID, al

    mov al, GLOBAL_UPDATE_ID
    cmp al, GLOBAL_INDEX
    jge UPDATE_INVALID_ID

    CALL UPDATE_VEHICLE_INFO
    jmp MAIN_MENU

UPDATE_INVALID_ID:
    PRINT_MSG invalid_update_id_msg
    PRINT_MSG retuen_main_menu
    jmp MAIN_MENU

; ============================================================
; FEATURE 4: CALCULATE PARKING FEE
; MENU HANDLER
; ============================================================
JUMP_CALCULATE_FEE:
    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h
    PRINT_MSG fee_header_msg
    PRINT_MSG fee_rate_msg
    PRINT_MSG fee_vehicle_prompt

    mov ah, 1
    int 21h
    sub al, 30h
    mov FEE_VEHICLE_ID, al

    mov al, FEE_VEHICLE_ID
    cmp al, GLOBAL_INDEX
    jge FEE_INVALID_ID

    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h
    PRINT_MSG fee_hours_prompt

    mov ah, 1
    int 21h
    sub al, 30h
    cmp al, 1
    jl FEE_INVALID_DURATION
    cmp al, 9
    jg FEE_INVALID_DURATION
    mov PARKING_HOURS, al

    CALL CALCULATE_PARKING_FEE
    jmp MAIN_MENU

FEE_INVALID_ID:
    PRINT_MSG invalid_id_msg
    PRINT_MSG retuen_main_menu
    jmp MAIN_MENU

FEE_INVALID_DURATION:
    PRINT_MSG fee_invalid_duration_msg
    PRINT_MSG retuen_main_menu
    jmp MAIN_MENU

; ============================================================
; FEATURE 3: SEARCH VEHICLE
; MENU HANDLER
; ============================================================
JUMP_SEARCH_VEHICLE:
    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h
    PRINT_MSG search_prompt_msg

    mov ah, 1
    int 21h
    sub al, 30h
    mov SEARCH_ID, al

    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h

    mov al, SEARCH_ID
    cmp al, GLOBAL_INDEX
    jge SEARCH_NOT_FOUND

    PRINT_MSG search_header_msg

    mov ah, 9
    lea dx, search_id_label
    int 21h

    mov ah, 9
    lea dx, search_name_label
    int 21h
    DISPLAY_VEHICLE_DETAILS_NAME_SEARCH SEARCH_ID

    mov ah, 9
    lea dx, search_loc_label
    int 21h
    DISPLAY_VEHICLE_DETAILS_PARKING_SLOT_SEARCH SEARCH_ID

    mov ah, 9
    lea dx, search_stat_label
    int 21h

    jmp SEARCH_DONE

SEARCH_NOT_FOUND:
    PRINT_MSG invalid_id_msg

SEARCH_DONE:

    PRINT_MSG retuen_main_menu

; FEATURE 5: REMOVE VEHICLE

; ============================================================

    mov ah, 2
    mov dx, 10
    int 21h

    int 21h

END:

MAIN ENDP

; ============================================================
; FEATURE 1: REGISTER A NEW VEHICLE
; MAIN PROCEDURE

    PRINT_MSG divider
    SET_ID
    TAKE_NAME_INPUT
    TAKE_PARKING_SLOT_INPUT
    SET_STATUS
    PRINT_MSG divider
    RET
ADD_VEHICLE ENDP

; ============================================================
; FEATURE 2: UPDATE VEHICLE INFORMATION
; MAIN PROCEDURE
; ============================================================
UPDATE_VEHICLE_INFO PROC
    PRINT_MSG divider
    PRINT_MSG update_options
    PRINT_MSG update_name_option
    PRINT_MSG update_parking_slot_option
    PRINT_MSG update_status_option

    RETRY:

    mov ah, 1
    int 21h
    sub al, 30h

    cmp al, 1
    je JUMP_UPDATE_NAME

    cmp al, 2
    je JUMP_UPDATE_PARKING_SLOT

    cmp al, 3
    je JUMP_UPDATE_STATUS

    jmp END_UPDATE_VEHICLE_INFO

    JUMP_UPDATE_NAME:
    UPDATE_NAME GLOBAL_UPDATE_ID
    jmp END_UPDATE_VEHICLE_INFO

    JUMP_UPDATE_PARKING_SLOT:
    UPDATE_PARKING_SLOT GLOBAL_UPDATE_ID
    jmp END_UPDATE_VEHICLE_INFO

    JUMP_UPDATE_STATUS:
    PRINT_MSG CHOSE_STATUS_UPDATE_OPTIONS
    PRINT_MSG CHOSE_STATUS_UPDATE_OPTIONS_1
    PRINT_MSG CHOSE_STATUS_UPDATE_OPTIONS_2
    mov ah, 1
    int 21h
    sub al, 30h

    UPDATE_STATUS GLOBAL_UPDATE_ID, al
    jmp END_UPDATE_VEHICLE_INFO

    END_UPDATE_VEHICLE_INFO:
    PRINT_MSG divider
    PRINT_MSG retuen_main_menu
    RET
UPDATE_VEHICLE_INFO ENDP

; ============================================================
; FEATURE 4: CALCULATE PARKING FEE
; MAIN PROCEDURE
; Calculates parking rate x parking duration.
; ============================================================
CALCULATE_PARKING_FEE PROC
    PRINT_MSG fee_total_msg

    mov al, PARKING_RATE
    mov bl, PARKING_HOURS
    mul bl

    mov bx, 10
    div bl

    mov bh, ah
    add al, 30h
    mov dl, al
    mov ah, 2
    int 21h

    mov al, bh
    add al, 30h
    mov dl, al
    mov ah, 2
    int 21h

    PRINT_MSG fee_currency_msg
    PRINT_MSG fee_complete_msg
    PRINT_MSG divider
    PRINT_MSG retuen_main_menu
    RET
CALCULATE_PARKING_FEE ENDP

; ============================================================
; FEATURE 5: REMOVE VEHICLE
; MAIN PROCEDURE
; Shifts later records left after removal.
; ============================================================
DELETE_VEHICLE PROC

    
    mov al, GLOBAL_INDEX
    sub al, GLOBAL_DELETE_ID
    sub al, 1
    mov DEL_SHIFT_COUNT, al      

    cmp al, 0
    je DELETE_SKIP_SHIFT         

 
    mov al, GLOBAL_DELETE_ID
    mov DEL_CUR_SLOT, al

DEL_SHIFT_LOOP:

    
    mov al, DEL_CUR_SLOT
    inc al                       
    mov ah, 00h
    mov si, ax
    mov dl, VEHICLE_ID[si]        

    mov al, DEL_CUR_SLOT
    mov ah, 00h
    mov si, ax
    mov VEHICLE_ID[si], dl        

    
 
    mov al, DEL_CUR_SLOT
    mov ah, 00h
    mov si, 20
    mul si                       
    mov di, ax                   

    mov al, DEL_CUR_SLOT
    inc al                       
    mov ah, 00h
    mov si, 20
    mul si                       
    mov si, ax                   

    mov bx, 20                   
    DEL_COPY_NAME:
        mov al, VEHICLE_OWNER[si]
        mov VEHICLE_OWNER[di], al
        inc si
        inc di
        dec bx
        jnz DEL_COPY_NAME

 
    mov al, DEL_CUR_SLOT
    mov ah, 00h
    mov si, 20
    mul si
    mov di, ax

    mov al, DEL_CUR_SLOT
    inc al
    mov ah, 00h
    mov si, 20
    mul si
    mov si, ax

    mov bx, 20
    DEL_COPY_LOC:
        mov al, PARKING_SLOT[si]
        mov PARKING_SLOT[di], al
        inc si
        inc di
        dec bx
        jnz DEL_COPY_LOC

    
    mov al, DEL_CUR_SLOT
    mov ah, 00h
    mov si, 20
    mul si
    mov di, ax

    mov al, DEL_CUR_SLOT
    inc al
    mov ah, 00h
    mov si, 20
    mul si
    mov si, ax

    mov bx, 20
    DEL_COPY_STAT:
        mov al, VEHICLE_STATUS[si]
        mov VEHICLE_STATUS[di], al
        inc si
        inc di
        dec bx
        jnz DEL_COPY_STAT

    
    inc DEL_CUR_SLOT             
    dec DEL_SHIFT_COUNT          
    jnz DEL_SHIFT_LOOP           

DELETE_SKIP_SHIFT:
    dec GLOBAL_INDEX

    
    mov al, GLOBAL_INDEX
    cmp al, 0
    je DELETE_RENUMBER_DONE

    mov DEL_CUR_SLOT, 0          
    DEL_RENUMBER_LOOP:
        mov al, DEL_CUR_SLOT
        mov ah, 00h
        mov si, ax
        add al, 48               
        mov VEHICLE_ID[si], al
        inc DEL_CUR_SLOT
        mov al, DEL_CUR_SLOT
        cmp al, GLOBAL_INDEX
        jl DEL_RENUMBER_LOOP
DELETE_RENUMBER_DONE:

    PRINT_MSG delete_ok_msg
    PRINT_MSG divider
    PRINT_MSG retuen_main_menu
    RET
DELETE_VEHICLE ENDP

STORAGE_CAPACITY_CHECK PROC
    PRINT_MSG cap_header_msg

    mov ah, 9
    lea dx, cap_current_msg
    int 21h

    mov al, GLOBAL_INDEX
    cmp al, 10
    je PRINT_TEN

    add al, 30h
    mov dl, al
    mov ah, 2
    int 21h
    jmp CAPACITY_NUMBER_DONE

PRINT_TEN:
    PRINT_MSG cap_ten_msg

    CAPACITY_NUMBER_DONE:

    mov ah, 2
    mov dx, 10
    int 21h
    mov dx, 13
    int 21h

    PRINT_MSG cap_max_msg

    mov al, GLOBAL_INDEX
    cmp al, MAX_CAPACITY
    jl  CAP_SPACE_OK
    PRINT_MSG cap_full_msg
    jmp CAP_CHECK_DONE

CAP_SPACE_OK:
    PRINT_MSG cap_ok_msg

CAP_CHECK_DONE:
    PRINT_MSG divider
    RET
STORAGE_CAPACITY_CHECK ENDP
